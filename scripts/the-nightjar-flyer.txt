# Murder on the Nightjar Flyer — a story script for the Chorus text engine.
#
# An overnight express pulled by an aether-locomotive, somewhere between
# London and Vienna, in an empire where every lamp hums and every door can
# be sealed with a voice. Lord Vane boarded with three trunks, a secretary,
# a wife, and a patent worth more than the train. He will not reach Vienna.
#
# The player is Inspector MARCH. The engine's whole twelve-verb core set
# (§4.3) carries the case, and the story exercises every section of the
# script format (§10) — this file is the engine's showcase:
#   READ / STUDY — study what the dead left behind (minds its languages)
#   TALK         — question the living            (talk to <who> about <idea>)
#   EXPLORE      — search the cars                (hidden things, once each)
#   USE_ON       — convene the inquest            (accuse with evidence in hand)
#   TRAVEL_TO /
#   LOCATE_AT    — the corridor, and standing still on it
#   WORK         — an honest shift below          (coin attr, FUNDS pool)
#   MANSUS       — the fire doorway, walked in pairs
#   DREAM        — sleep on an idea               (the DREAMS deck, ADVANCE)
#   ACQUIRE      — sign for the claim ticket      (a cost, and a MOVE)
#   WAIT         — let the timetable speak        (the train acts on the clock)
#
# The timetable is the culprit's alibi: evidence burns at 2:00, the telegram
# only lands at the border at midnight, and Vienna scatters everyone at 4:00.
# The clock ticks in halfhours — 0:42 is 21:00 — and the triggers below show
# every authored form: `once` and recurring, `at halfhour N` and the `at
# hour N` alias (§10), and the expiry form `on expire of NAME` (§6.6)
# beside the built-in destroy.
#
# Suspicion rides the night with you. SUSPICION is a pool, not a card: every
# careless search spawns a copy, and the conductor's rounds (22:30, every
# night he works) spawn more when his roll goes against you. Only two things
# on this train happen more than once — his rounds and the dawn watch. Three
# copies in his book and watch puts the case, and you, off the Flyer: the
# third copy lands the escalation ending, wherever in the night it falls.
# The passengers trade gossip from the RUMOURS deck, the rude searches are
# rolled against the searcher's nerve, and dreaming draws from the DREAMS
# deck — a second deck with neither `defaultcard` nor `reset`, whose
# exhaustion is a loud failure (§4.4).
#
# seed: 1899 — calibrated for the scripted win path: the cabin search's
# patrol and the 22:30 rounds both
# land the conductor's middle tier, so two SUSPICION gather in his book — the
# pool exists, the escalation never trips — and the firebox ash gives a clean
# find. Change it and the night plays differently.

script: Murder on the Nightjar Flyer
player: MARCH
goal: unmask Lord Vane's killer — gather proofs aboard and put them to the inquest before Vienna
start: 0:42
seed: 1899

# ── the timetable speaks ─────────────────────────────────────────────────
# body_found hands the late lord a `expires` clock: one day, then the
# authored `on expire` trigger replaces him (REPLACE, §4.4) instead of the
# built-in destroy. The tonic glass takes the built-in path: SET with a
# quoted duration (§4.2 — the quoted path accepts every §5.1 unit, the
# halfhour included) and no authored trigger. vienna_arrival is the
# deadline: reach Vienna with no name on the tally-board and the story
# quietly ends (ENDING in a trigger, the default flavour, §4.4).
triggers:
  body_found once at halfhour 44 -> DESTROY THE_PATENTEE, SPAWN THE_LATE_LORD INTO FIRST_CABIN, SPAWN THE_AETHER_SEAL INTO FIRST_CABIN, SET(FIRST_CABIN, unsealed, 1), SET(THE_AETHER_SEAL, state.sealed, true), SET(THE_LATE_LORD, expires, 1d), NARRATE "A cry from the first cabin. Lord Vane lies behind his own aether-seal, unrobbed, unwounded, and entirely dead."
  night_porter once at halfhour 46 -> night_watchman()
  conductor_rounds at halfhour 45 -> DRAW RUMOURS INTO GUARD_VAN, patrol_check(0)
  border_stop once at halfhour 0 -> SPAWN THE_TELEGRAM INTO GUARD_VAN, NARRATE "The Flyer hisses down at the border. A telegraph boy sprints the platform and puts a French telegram into Brack's white glove."
  the_long_tunnel once at hour 1 -> UNRELATE(FIREBOX, GUARD_VAN, south), NARRATE "The train enters the Vendahl tunnel. For three hours the lamps burn blue, every mirror fogs over — and the crew door to the firebox closes like a held breath."
  the_furnace_takes_it once at halfhour 4 -> DESTROY THE_VANE_LEDGER, NARRATE "A thin smell of ash reaches the corridor: someone is feeding the firebox, and a ledger cover stamped VANE is curling in the flames."
  dawn_watch at hour 4 -> RELATE(FIREBOX, GUARD_VAN, south), NARRATE "The small hours turn. The Vendahl tunnel lets go of the train, and the crew door to the firebox breathes again."
  lord_tended on expire of THE_LATE_LORD -> REPLACE THE_LATE_LORD WITH THE_TENDED_LORD, NARRATE "The company's people come for the late lord at last, and carry him, tended and tagged, forward through the cars."
  vienna_arrival once at halfhour 12 -> NARRATE "Vienna lights slide along the glass, and the Flyer glides in with her mystery intact — the passengers scatter, and the case scatters with them.", ENDING QUIET_NIGHT

# ── the company's book on the inspector ──────────────────────────────────
# sus spawns one SUSPICION onto the named card and tells the corridor about
# it; watch is the escalation — three copies in his book and the case leaves
# the train; patrol_check is the conductor's rounds, a roll against his mood
# that calls watch. night_watchman takes no parameters at all.
functions:
  sus who -> SPAWN SUSPICION INTO who, NARRATE "A steward will remember {who} asking after things that are none of an inspector's business."
  watch(n) -> IF count(name:SUSPICION) >= n -> (ENDING OFF_THE_FLYER)
  patrol_check(dc) -> ROLL(2, 6, dc), watch(3), IF roll >= 8 -> (NARRATE "The conductor's rounds take {roll} minutes and pass you by — two taps on the door, a bored hum on the downslope.") ELSE IF roll >= 5 -> (sus(MARCH)) ELSE -> (NARRATE "He walks the corridor half-asleep, {roll} minutes of boots on carpet, and the night stays quiet.")
  night_watchman -> NARRATE "The corridor lamps dim to embers. The porter walks the train from end to end, counting doors under his breath."

# ── what the corridor gossips and dreams about ───────────────────────────
# the draw order is the seed-fixed permutation of the spec (2:1:1 weighting);
# DREAMS shows the bare deck line — no defaultcard, no reset (§4.4)
decks:
  RUMOURS spec RUMOUR, RUMOUR, HEARSAY, SCANDAL defaultcard RUMOUR reset
  DREAMS spec PORTENT, PORTENT, VOID

# ── how it can all end ───────────────────────────────────────────────────
# three table forms: an explicit melancholy, a grand, and the default
# flavour (melancholy, §10) — and all three can fire: watch ends it at
# three SUSPICION, the inquest ends it in glory, and if Vienna arrives
# before the tally-board does, QUIET_NIGHT takes the story unawares.
endings:
  OFF_THE_FLYER: "Off the Flyer at Speed" (melancholy)
  CASE_CLOSED: "Confession Before Vienna" (grand)
  QUIET_NIGHT: "The Flyer Reaches Vienna"

# ── what can be done ─────────────────────────────────────────────────────
# the twelve core verbs (§4.3), every slot count form (§6.1), every query
# term (§4.5), and every outcome op (§4.4) somewhere in the lists below.
verbs:
  READ:
    actors: 1..2 kind:character
    object: 1 kind:text
    at: 0..1 kind:place
    duration: 1h
    recipes:
      read.manifest when object.aspect.manifest >= 1 -> NARRATE "The passenger manifest, first class: the lord Vane, his wife Verity, a secretary named Quill, a spiritualist trading as Seraphine. Crew of two on the sheet — a guard and a stoker. Six strangers and a dead man, once the cry goes up."
      read.french_gate hidden when object.aspect.language.french >= 1 AND NOT any(actors, aspect:language.french >= 1) -> REFUSE("The words are French to you, and your French died at school.")
      read.cylinder when object.aspect.tome.cylinder >= 1 -> PROGRESS(20m, "You wind the cylinder onto the phonomotor; the stylus trembles and finds its groove."), PROGRESS(50m, "A voice assembles out of the hiss — the lord's own, speaking the words that open his seal."), SPAWN THE_STOLEN_VOICE, SET(THE_AETHER_SEAL, state.sealed, false), DESTROY OBJECT, NARRATE "{actors} play the cylinder to its end. Someone owns the lord's voice, and the seal answers it."
      read.telegram when object.aspect.tome.telegram >= 1 -> PROGRESS(1halfhour, "Seraphine's lips move; the French unbraids itself line by line."), SPAWN THE_VIENNA_WARNING, DESTROY OBJECT, NARRATE "\"Tell Q he signs nothing after Vienna — not the patents, not the funeral arrangements.\" The partner knew. Did Q?"
      read.ledger when object.aspect.tome.ledger >= 1 -> PROGRESS(1halfhour, "Two hands wrote this ledger, and in the margins they argue."), SPAWN THE_EMBEZZLEMENT, HUNCH, DESTROY OBJECT, NARRATE "Quill's neat corrections hide a steady drain on the aether royalties. The lord had found it. Quill knew he had — and the shape of it stays with you."
      read.letter when object.aspect.tome.letter >= 1 -> SPAWN THE_DISINHERITANCE, DESTROY OBJECT, NARRATE "The solicitor begs the lord to come home at once and rewrite his will before the Vienna signing. Whoever marries grief gets nothing; whoever holds the pen gets everything."
      read.companions when count(actors) = 2 AND rel(neighbor, actors) -> NARRATE "{actors} read over each other's shoulders, and the book warms between them."
      default -> NARRATE "{actors} read {object}."

  TALK:
    actors: 1 kind:character
    to: 1 kind:character
    about: 0..1 kind:idea
    duration: 1halfhour
    recipes:
      talk.meet_guard when to.aspect.unmet >= 1 AND to.aspect.duty >= 1 -> DESTROY THE_GUARD, SPAWN BRACK INTO GUARD_VAN, RELATE(MARCH, BRACK, neighbor, 2), NARRATE "\"Brack. Railway security,\" he says, and the white gloves settle. \"They wired ahead about an inspector. The company seal is in my van — and somebody's hands have been on it.\""
      talk.meet_widow when to.aspect.unmet >= 1 AND to.aspect.grief >= 1 -> DESTROY THE_WIDOW, SPAWN VERITY INTO DINING_CAR, NARRATE "\"Verity Vane,\" she says, mourning jet buttoned to the throat. \"You knew the name before I said it. Everyone on this train knows everything by now.\""
      talk.meet_secretary when to.aspect.unmet >= 1 AND to.aspect.secret >= 1 -> DESTROY THE_SECRETARY, SPAWN QUILL INTO DINING_CAR, NARRATE "\"Quill — the lord's secretary,\" he says, and presents his alibi in triplicate. \"I was in the dining car the whole evening. Ask anyone. Please.\""
      talk.meet_stoker when to.aspect.unmet >= 1 AND to.aspect.steam >= 1 -> DESTROY THE_STOKER, SPAWN COLE INTO FIREBOX, NARRATE "\"Cole. I drive the firebox, and the firebox drives me,\" the stoker says. \"You want the timetable, inspector. The timetable is the only witness that never lies.\""
      talk.meet_spiritualist when to.aspect.unmet >= 1 AND to.aspect.language.french >= 1 -> DESTROY THE_SPIRITUALIST, SPAWN SERAPHINE INTO DINING_CAR, DRAW RUMOURS INTO DINING_CAR, IF exists(THE_AETHER_SEAL) -> (NARRATE "\"Seraphine reads the aether for the first-class trade,\" she says, turning a card face down. \"Mind — I read what is written, not what is guilty. Though tonight we all read the same card: the lord is dead.\"") ELSE -> (NARRATE "\"Seraphine reads the aether for the first-class trade,\" she says, turning a card face down. \"Mind — I read what is written, not what is guilty.\"")
      talk.meet_lord when to.aspect.unmet >= 1 AND to.aspect.patron >= 1 -> DESTROY THE_PATENTEE, SPAWN LORD_VANE INTO FIRST_CABIN, NARRATE "\"Vane — patents, port, and the Vienna signing,\" the lord says, sealing his own cabin door behind him. \"A man in my position trusts nothing he cannot own.\""
      talk.seraphine_seal when to.aspect.aether >= 2 AND about.aspect.case.seal >= 1 -> NARRATE "\"An aether-seal answers one voice only,\" Seraphine says. \"If the lord did not open it himself, then somebody carried his voice in their pocket.\""
      talk.seraphine_default when to.aspect.aether >= 2 -> NARRATE "Seraphine turns a card, glances at it, and lays it face down. \"The dead speak through steam and cards alike,\" she says. \"It is the living who mumble.\""
      talk.cole_seal when to.aspect.steam >= 2 AND about.aspect.case.seal >= 1 -> NARRATE "Cole wipes his hands on a rag that is dirtier than they are. \"Someone paid me to rake the firebox hot at midnight,\" he mutters. \"I kept the ash, though. Ash keeps secrets — and {about} is the secret it keeps.\""
      talk.cole_default when to.aspect.steam >= 2 -> NARRATE "Cole explains boiler pressures until you can politely escape."
      talk.verity_will when to.aspect.grief >= 1 AND about.aspect.case.will >= 1 -> NARRATE "\"He meant to change it. Yes.\" Verity's mourning jet is buttoned to the throat. \"You would grieve differently, Inspector, if your marriage were a clause.\""
      talk.verity_default when to.aspect.grief >= 1 -> NARRATE "Verity weeps with flawless technique and answers nothing at all."
      talk.brack_telegram when to.aspect.duty >= 2 AND about.aspect.case.chance >= 1 -> NARRATE "\"A man in uniform holds what he is handed until it is claimed,\" Brack says, and the silence after it is the longest sentence he has ever spoken."
      talk.brack_default when to.aspect.duty >= 2 -> NARRATE "Brack recites the company regulations concerning death aboard, verbatim, twice."
      talk.quill_motive when to.aspect.secret >= 2 AND about.aspect.case.motive >= 1 -> NARRATE "\"The lord was careless with figures,\" Quill says. His cufflinks catch the light. His voice does not."
      talk.quill_default when to.aspect.secret >= 2 -> NARRATE "Quill apologizes for everything except his alibi, which he presents in triplicate."
      talk.vane_default when to.aspect.patron >= 1 -> NARRATE "The lord talks patents and port. Behind him the seal on his cabin door breathes like a sleeping animal."
      default -> NARRATE "{actors} exchange words with {to}."

  EXPLORE:
    actors: 1 kind:character
    object: 1 kind:place
    duration: 1h
    recipes:
      explore.cabin when object.aspect.site.first >= 1 AND object.attr.unsealed >= 1 AND NOT object.attr.searched >= 1 -> PROGRESS(1halfhour, "The seal parts for your warrant with a sound like a struck match."), SPAWN THE_SOLICITOR_LETTER INTO OBJECT, SET(OBJECT, searched, 1), patrol_check(1), NARRATE "The cabin is orderly the way a stage set is orderly. Behind the writing case: a solicitor's letter — and the Vane household ledger, far from its shelf."
      explore.firebox when object.aspect.site.firebox >= 1 AND NOT object.attr.searched >= 1 -> ROLL(total(actors, aspect:nerve), 6), IF roll >= 5 -> (PROGRESS(1halfhour, "You sift warm ash while Cole watches the shovel a shade too carefully."), SPAWN THE_WAX_CYLINDER INTO OBJECT, SET(OBJECT, searched, 1), NARRATE "{roll} minutes of sifting, well spent: a wax phonomotor cylinder, half-melted but intact, the lord's monogram on the brass case.") ELSE IF roll <= 3 -> (SET(OBJECT, searched, 1), sus(MARCH), NARRATE "You rummage loudly enough that Cole forgets the shovel. {roll} minutes, nothing to show.") ELSE -> (SET(OBJECT, searched, 1), NARRATE "Ash, clinker, and {roll} minutes of your night gone. The firebox keeps what it keeps.")
      explore.dining when object.aspect.site.dining >= 1 AND NOT object.attr.searched >= 1 -> PROGRESS(1halfhour, "Under the Mahrajah's table, among the broken biscuits."), SPAWN THE_TONIC_GLASS INTO OBJECT, SET(THE_TONIC_GLASS, expires, "2halfhour"), SET(OBJECT, searched, 1), NARRATE "The lord's table still stands. His tonic glass smells wrong — aether where the laudanum should be, and not long for this world."
      explore.luggage when object.aspect.site.luggage >= 1 AND NOT object.attr.searched >= 1 -> ROLL(total(actors, aspect:nerve), 6), IF roll > 8 -> (PROGRESS(1halfhour, "Trunk after trunk, manifest after manifest."), SPAWN THE_STOPPED_WATCH INTO OBJECT, REVEAL THE_STOPPED_WATCH, SET(OBJECT, searched, 1), NARRATE "{roll} minutes of careful work, and there it is between the trunks: a gentleman's pocket watch, stopped at 11:50, still warm.") ELSE IF roll >= 6 -> (SET(OBJECT, searched, 1), NARRATE "{roll} minutes among other people's luggage, and nothing opens quietly for you tonight.") ELSE -> (SET(OBJECT, searched, 1), sus(MARCH), NARRATE "You pry a trunk that is not yours — {roll} minutes of noise, and the luggage van remembers.")
      explore.guard when object.aspect.site.guard >= 1 AND NOT object.attr.searched >= 1 -> PROGRESS(1halfhour, "Brake-dust, ledger-oil, and Brack's spine straightening by the inch."), SET(OBJECT, searched, 1), NARRATE "Brack's tally-board is spotless. Nothing on a night train is spotless."
      explore.searched when object.attr.searched >= 1 AND exists(kind:character aspect:duty>=1 -name:MARCH) -> sus(MARCH), watch(3), NARRATE "{actors} turn the place over again and find nothing new — only an audience."
      explore.cabin_sealed when object.aspect.site.first >= 1 -> NARRATE "The aether-seal breathes cold against your warrant. The door does not open for a piece of paper."
      default -> NARRATE "There is nothing here to search."

  USE_ON:
    actors: 1 kind:character
    object: 1 kind:rite
    with: 0..3 kind:idea
    duration: 2h
    recipes:
      inquest.short shown hint "The tally-board wants three proofs laid together — pooled, not promised." when count(with) >= 1 AND total(with, aspect:proof) < object.aspect.demand -> REFUSE("The van hears you out and finds the case wobbles. Nobody is held on a hunch.")
      inquest.win when total(with, aspect:proof) >= 3 AND every(with, aspect:proof >= 1) -> PROGRESS(1halfhour, "Brack convenes the van; the company seal goes on the door and the words go on the record."), PROGRESS(1h30m, "Motive, means, opportunity: {with} — every card on the tally-board points the same way down the corridor."), NARRATE "Embezzled royalties, a stolen voice for the seal, a telegram that never reached its addressee — the cuffs unpin Quill where the constable cannot. He confesses before Vienna.", SPAWN CONFESSION_OF_QUILL, REPLACE OBJECT WITH THE_CLOSED_TALLY, ENDING CASE_CLOSED
      default -> REFUSE("You would convene an inquest with empty hands. Brack pretends not to hear.")

  TRAVEL_TO:
    actors: 1 kind:character
    object: 1 kind:place
    duration: 1halfhour
    recipes:

  WAIT:
    actors: 1 kind:character
    duration: 1h
    recipes:

  STUDY:
    # READ's signature (§4.3) at a compound duration (§5.1); the french gate
    # carries no id — the compiler mints study.recipe0 — and is POSTED:
    # shown, with a hint (§6.2, §6.4)
    actors: 1..2 kind:character
    object: 1 kind:text
    at: 0..1 kind:place
    duration: 1h30m
    recipes:

      default -> NARRATE "{actors} study {object}, line by line, and the lines hold still for it."

  WORK:
    # the wages verb: an open-ended actor count (§6.1) and a slot query with a
    # negated aspect term (§4.5) — no day-labor in the sealed first cabin
    actors: 2.. kind:character
    at: 0..1 kind:place -aspect:site.first
    duration: 2h
    recipes:
      work.firebox when at.aspect.site.firebox >= 1 -> SET(ACTORS, coin, +2), SPAWN resource:FUNDS, NARRATE "{actors} take the watch below at {at} — coal, steam, and two coins a head.", IF count(name:FUNDS) > 3 -> (NARRATE "The paymaster eyes the coal bill and says nothing.")
      work.owed when actors.attr.coin != 0 -> SET(ACTORS, coin, +1), NARRATE "The paymaster adds another coin to what the company already owes {actors}."
      default -> SET(ACTORS, coin, +1), NARRATE "{actors} put in an honest shift, and the company's book grows one coin longer."

  MANSUS:
    # the crossing is walked in pairs — one to knock, one to hold the lamp —
    # an exact slot count (§6.1); the fire-door rolls with the place's own
    # aspect as bonus (§4.4) and nests an IF inside an IF (§6.8)
    actors: 2 kind:character
    object: 1 kind:place
    duration: 1halfhour
    recipes:
      mansus.firebox when object.aspect.site.firebox >= 1 -> ROLL(1, 6, object.aspect.site.firebox), IF roll >= 4 -> (IF roll = 6 -> (SET(ACTORS, aspect:nerve, +1), NARRATE "The fire-door breathes like a held note, and {actors} breathe back. The glare takes a measure of doubt and pays it in nerve.") ELSE -> (NARRATE "The fire stares, and for a moment the fire is a doorway. {actors} look away first.")) ELSE -> (NARRATE "The fire is only fire tonight. Some doors do not answer.")
      default -> NARRATE "{actors} look long at the place, and the place looks back."

  LOCATE_AT:
    # the utility verb — a sub-halfhour duration leaves dead air behind (§6.6)
    actors: 1 kind:character
    object: 1 kind:place
    duration: 20m
    recipes:
      default -> MOVE ACTORS INTO OBJECT, NARRATE "{actors} take up a station at {object} and stay put."

  DREAM:
    # an open-ended slot count with a three-term query — kind, aspect level,
    # and the state flag the seal carries until its voice is played (§4.5);
    # the taken branch draws from the DREAMS deck (no INTO) and ADVANCEs the
    # window (§6.6)
    actors: 1 kind:character
    with: 0.. kind:idea aspect:secret>=1 state:!sealed
    duration: 1h
    recipes:
      dream.seal when any(with, aspect:case.seal >= 1) -> DRAW DREAMS, ADVANCE(30m), NARRATE "{actors} sleep on the seal and wake with its shape behind the eyes. The dream leaves a card on the pillow."
      default -> NARRATE "{actors} sleep without dreaming, and the night files it away."

  ACQUIRE:
    # the cost verb — a negative delta may go below zero (§4.4), and the
    # canonical paren'd MOVE carries the ticket into the signer's pocket;
    # the signed flag keeps the company from charging twice for one ticket
    actors: 1 kind:character
    object: 1 kind:item
    duration: 30m
    recipes:
      acquire.claim when object.aspect.claim >= 1 AND NOT object.attr.signed >= 1 -> SET(ACTORS, coin, -1), MOVE(OBJECT, INTO ACTORS), SET(OBJECT, signed, 1), NARRATE "{actors} sign for the claim ticket, and the company takes its pound of flesh in paperwork."
      default -> REFUSE("Nobody aboard is selling.")
