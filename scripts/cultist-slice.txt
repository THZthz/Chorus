# The Bellwood Aspirant — a Cultist-Simulator vertical slice for the Chorus text engine.
#
# Hollowmere: a fog-bound cathedral town where the bells ring one note too many
# and the fog keeps whatever the bells say. The player is WREN, a restorer of
# old ledgers with three foundations (VIGOR, WITS, YEARNING), no silver, two
# thin books, and a rite she has no business owning. The design doc for this
# slice is docs/plans/v0.6-cs-slice.md; the verified mechanics it authorizes are
# docs/reverse/cultist-simulator.md §2/§3/§5/§6. This file is the COMPILING
# SKELETON: the full card inventory, the verbs and their slots, the decks, the
# endings, and the card-borne machines (the labor rolls, the whisper decay, the
# wheel's bootstrap). The 101 recipes are the authoring phases' work (the doc's
# §4 inventories every one) — each verb below carries only its default, or the
# one stub another section's load check names, so the file loads today.
#
#   WORK      — labor and the clerk's post    (silver, the labor rolls, the bootstrap)
#   WAIT      — the wheel                     (THE_TURNING: the season engine, self-rechaining)
#   STUDY     — the lore ladder               (radiance and hush, three rungs each, the Way)
#   DREAM     — moods and the desire chain    (the MOODS deck, promise → vow → hunger)
#   TALK      — the congregation              (Mercy Hale: acquaintance → believer → disciple)
#   MANSUS    — the Bellwood crossing         (one door, one deck, no reset)
#   USE_ON    — the Lifted Lamp               (one rite: one summon, the desire ascension)
#   ACQUIRE   — the flea market               (the skeleton key, tin candles, grey salve)
#   TRAVEL_TO /
#   LOCATE_AT — the two streets the slice owns
#
# Ethics (hard constraint, docs/reverse/cultist-simulator.md ethics note): every
# label, narration and ending in this file is fresh text written for the slice.
# Text policy (owner, 2026-10-16): CS strings may be quoted for this
# engine-validation slice (open-source, non-commercial, game owned); the
# Hollowmere frame stays. Mechanics/numbers remain dossier-governed facts.
# structure are reused. The principles (RADIANCE, HUSH), the door (the
# Bellwood), the rite (the Lifted Lamp), the summon (the Tolling Guest), the
# desire (the Hunger of Heights), the people and every sentence are this repo's
# own.
#
# Two engine laws this file obeys (both verified by run, 2026-10-15):
#   1. REPLACE's successor resolves strictly inside trigger bodies — a never-
#      alive prototype there bricks every commit on the verb — so every
#      card-borne transform is DESTROY + SPAWN + SET. Recipes may REPLACE
#      into prototypes freely (the mint carries the prototype's aspects).
#   2. Predicates read aspects and attrs, never names: every card a recipe
#      must recognize inside a bound role carries a one-level marker aspect
#      (weary, dim, pale, dread, rapture, ease, mending, book, fever, frail,
#      toll, ...). WAKEFULNESS and GLIMMER stay bare — nothing binds them.
# The market's stock is finite: what the trestles hold at the start is all
# Hollowmere has (a restock trigger would have to mint a declared entity,
# which the engine refuses — the brick this file once shipped).
#
# The wheel is the slice's engine-from-a-single-spark. The first completion that
# binds a job card (VIGOR, the clerk's post) fires its `on complete job` clause:
# spawn THE_TURNING if absent, then RUN "time.turn" cross-verb (§9.48 — CS's
# job→needs induce). time.turn drains one silver, draws a weather from the
# SEASONS deck, and hands the cycle to that weather's recipe; every weather ends
# by running time.turn again — the dossier's needs → discardseason → drawseason
# → needs loop, as RUN chains (§9.47). Later bootstraps find THE_TURNING alive,
# spawn nothing, and are suppressed exactly as CS's are.
#
# Time scale: 1 CS second → 1 engine minute (a 60 s action is the 1 h verb
# duration; a 60 s season cycle is the 1 h WAIT window; the summon's 180 s is
# the guest's 3 h). Chances are CS-exact: 30% labor and induction rolls, the
# 90/70/30 cult-business convention, 70% hunt and cure rolls, 90% ascension
# (Grand-rite 10% failure), the 3-mark death rules.
#
# seed: 1837 — chosen so the pinned win path's BELLWOOD draws run THE_LOST_BELL
# first (draw 1) and THE_BELL_HEART by draw 6, closing radiance 24 as 18 + 6;
# the dice calibrate in the authoring phases (docs/plans/v0.6-cs-slice.md §8).

script: The Bellwood Aspirant

# ── the card-borne clocks ────────────────────────────────────────────────
# every lifetime is SET x, expires at spawn; every transform is DESTROY +
# SPAWN + SET (the first engine law: no REPLACE into runtime prototypes from
# a trigger body — the commit trial resolves the successor strictly).
# WAKEFULNESS never fades: the chain ends there (simplification, documented).
triggers:
  toll_fades on expire of THE_FAR_TOLL -> DESTROY THE_FAR_TOLL, SPAWN THE_FAINT_TOLL, SET(THE_FAINT_TOLL, expires, 1h), NARRATE "The far toll thins to a faint one, the way a bell keeps ringing after the rope has gone still."
  faint_turns on expire of THE_FAINT_TOLL -> DESTROY THE_FAINT_TOLL, SPAWN WAKEFULNESS, NARRATE "The faint toll settles in the walls and will not finish. You will lie awake counting a bell that is no longer ringing."
  fever_sours on expire of THE_FEVER -> DESTROY THE_FEVER, SPAWN THE_FRAILTY, NARRATE "The fever breaks the wrong way. What it leaves is not health; it is an arrangement with weakness."
  guest_fades on expire of THE_TOLLING_GUEST -> IF NOT exists(name:THE_FAR_TOLL) -> (DESTROY THE_TOLLING_GUEST, SPAWN THE_FAR_TOLL, SET(THE_FAR_TOLL, expires, 1h), NARRATE "The guest thins into its own voice, and the voice into a far toll, and the fog files both.") ELSE -> (DESTROY THE_TOLLING_GUEST, NARRATE "The guest thins into its own voice, and the fog — which has heard it before — keeps only the habit of listening.")

# ── what can be done ─────────────────────────────────────────────────────
# the slice's nine verbs. Slot queries use the §9.49 alternation wherever CS's
# any-of slots demanded it (WORK's entry, the rite's invocation). Recipes are
# the authoring phases' work (docs/plans/v0.6-cs-slice.md §4 — 102 of them);
# each verb below carries its default, and WAIT carries the time.turn stub the
# bootstrap's RUN names at load.
verbs:
  WORK:
    # CS's one big any-of entry slot, as alternation: a foundation (intact or
    # drained — the marker aspects), the guest (uncanny labor), or the
    # frailty (ruined labor)
    actors: 1 kind:character
    with: 0..1 kind:resource aspect:vigor>=1 | kind:resource aspect:wits>=1 | kind:resource aspect:yearning>=1 | kind:resource aspect:weary>=1 | kind:resource aspect:dim>=1 | kind:resource aspect:pale>=1 | kind:character aspect:bell_voice>=1 | kind:state aspect:frail>=1
    at: 0..1 kind:place
    duration: 1h
    recipes:
      work.default -> NARRATE "{actors} put in an hour, and the hour gives back what it will."

  WAIT:
    # the wheel. time.turn is a stub until the season authoring lands; it must
    # exist now because the bootstrap's RUN names it (§9.47 load check)
    actors: 1 kind:character
    duration: 1h
    recipes:
      time.turn when exists(name:THE_TURNING) -> NARRATE "The wheel turns; the weathers are still being cut."
      default -> NARRATE "{actors} let the fog pass through the room."

  STUDY:
    actors: 1 kind:character
    object: 1 kind:text | kind:idea
    with: 0..2 kind:idea
    at: 0..1 kind:place
    duration: 2h
    recipes:
      study.default -> NARRATE "{actors} study {object}, line by line, and the lines mostly hold still."

  DREAM:
    actors: 1 kind:character
    with: 0..2 kind:state | kind:idea | kind:resource
    duration: 1h
    recipes:
      dream.default -> NARRATE "{actors} sleep without dreaming, and the fog files it away."

  TALK:
    actors: 1 kind:character
    to: 0..1 kind:character
    with: 0..2 kind:idea
    about: 0..1 kind:idea
    duration: 1halfhour
    recipes:
      talk.default -> NARRATE "{actors} exchange words with {to}, and the fog takes its customary share."

  MANSUS:
    actors: 1 kind:character
    object: 1 kind:place
    with: 0..1 kind:resource aspect:yearning>=1 | kind:character aspect:bell_voice>=1
    duration: 1h
    recipes:
      mansus.default -> REFUSE("The crossing does not open on a whim.")

    object: 1 kind:rite
    with: 0..3 kind:idea | kind:character aspect:bell_voice>=1 | kind:state aspect:fever>=1
    about: 0..1 kind:item
    to: 0..1 kind:idea aspect:vow>=1 | kind:idea aspect:desire>=1
    duration: 2h
    recipes:
      rite.default -> REFUSE("The lamp will not lift on empty hands.")

  ACQUIRE:
    actors: 1 kind:character
    object: 1 kind:item
    duration: 1halfhour
