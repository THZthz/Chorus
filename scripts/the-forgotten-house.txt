# The Forgotten House — a sample game script for the Chorus text engine.
# A piece of text IS the game: world, logic, and the table you play on.

script: The Forgotten House
player: ANA
start: 0:14

# ── the world ────────────────────────────────────────────────────────────
entities:
  HOUSE is place
  STUDY is place in HOUSE with study:1
  DINING_HALL is place in HOUSE
  VILLA is place
  GREENHOUSE is place in VILLA
  ANA is character with occult:2, lantern:1 at HOUSE
  BERTHA is character at HOUSE
  MY_FOLLOWER is character with lantern:2 at HOUSE
  THE_FORGETTING is text with secret:4 at HOUSE
  THE_PALE_LADY is text with secret:3, language.latin:2 at STUDY
  FAIRY_TALES is text with tender:1 at HOUSE
  MAGICK_BOOK is text with magick:1 at STUDY
  RITE_OF_ASCENSION is rite with demand:3 at HOUSE
  CHALICE is item at HOUSE
  CLUE_SECRET_HIDEOUT is idea with secret:1 at HOUSE
  ANA is parent_child of BERTHA

# ── cards that can come into being ───────────────────────────────────────
prototypes:
  CLUE is idea deriving name from object, secret from object.secret-2
  CLUE_SECRET_HIDEOUT is idea with secret:1
  PLAN_INFILTRATION is idea
  BONDING_MEMORY is idea
  MARK_OF_ASCENSION is idea
  DREAD is state

# ── the clock speaks ─────────────────────────────────────────────────────
triggers:
  dusk_falls at hour 16 -> SPAWN DREAD, NARRATE "The lamps come on."

# ── what can be done ─────────────────────────────────────────────────────
verbs:
  READ:
    actors: 1..2 kind:character
    object: 1 kind:text
    at: 0..1 kind:place
    duration: 1h
    recipes:
      read.magick_gate hidden when object.aspect.magick >= 1 AND NOT any(actors, aspect:occult >= 2) -> REFUSE("The letters swim like minnows.")
      read.parent_child when count(actors) = 2 AND rel(parent_child, actors) -> SPAWN BONDING_MEMORY, NARRATE "{actors} read aloud; the child sleeps wiser."
      read.unrelated_pair when count(actors) = 2 -> NARRATE "{actors} trade the book back and forth and lose the thread."
      read.secret when object.aspect.secret >= 3 -> SPAWN CLUE, NARRATE "{actors} tease out a hidden thread."
      default -> NARRATE "{actors} read {object}."

  TALK:
    actors: 1..3 kind:character
    about: 0..1 kind:idea
    at: 0..1 kind:place
    duration: 1h
    recipes:
      talk.plan when about.aspect.secret >= 1 -> SPAWN PLAN_INFILTRATION
      default -> NARRATE "{actors} talk."

  TRAVEL_TO:
    actors: 1..3 kind:character
    object: 1 kind:place
    duration: 1h
    recipes:
      default -> MOVE ACTORS INTO OBJECT, NARRATE "{actors} travel to {object}."

  USE_ON:
    actors: 1..5 kind:character
    object: 1 kind:rite
    with: 0..2 kind:item
    at: 0..1 kind:place
    duration: 3h
    recipes:
      rite.lantern_gate shown hint "The rite wants pooled lantern — it will starve." when total(actors, aspect:lantern) < object.aspect.demand -> REFUSE("The rite starves for light.")
      default -> DESTROY OBJECT, DESTROY WITH, SPAWN MARK_OF_ASCENSION INTO at

  WAIT:
    actors: 1 kind:character
    duration: 1h
    recipes:
      default -> NARRATE "{actors} wait."

  WORK:
    actors: 1 kind:character
    at: 0..1 kind:place
    duration: 2h
    recipes:
      default -> NARRATE "{actors} work at {at}."

  EXPLORE:
    actors: 1 kind:character
    object: 1 kind:place
    duration: 1h
    recipes:
      default -> NARRATE "{actors} explore {object}."

  DREAM:
    actors: 1 kind:character
    with: 0..1 kind:idea
    duration: 1h
    recipes:
      default -> NARRATE "{actors} dream."

  LOCATE_AT:
    actors: 1..3 kind:character
    object: 1 kind:place
    duration: 0m
    recipes:
      default -> MOVE ACTORS INTO OBJECT

  ACQUIRE:
    actors: 1 kind:character
    object: 1
    duration: 0m
    recipes:
      default -> NARRATE "{actors} acquire {object}."

# ── how it reads aloud ───────────────────────────────────────────────────
templates:
  READ says "{actors} read {object} at {at}."
  TALK says "{actors} talk about {about} at {at}."
  TRAVEL_TO says "{actors} travel to {object}."
  USE_ON says "{actors} perform {object} at {at}."
  WAIT says "{actors} wait."
  WORK says "{actors} work at {at}."
  EXPLORE says "{actors} explore {object}."
  DREAM says "{actors} dream."
  LOCATE_AT says "{actors} are at {object}."
  ACQUIRE says "{actors} acquire {object}."
